The fits_tools.quality_control module provides automated quality assessment for CCD/CMOS images. It combines validation checks with statistical analysis to grade images and flag potential problems.

Assessing a Single Image

assess_quality

from fits_tools.quality_control import assess_quality

metrics = assess_quality("image.fits")

print(f"Grade:   {metrics.quality_grade}")
print(f"Score:   {metrics.quality_score:.1f}/100")
print(f"Passed:  {metrics.passed}")
print(f"Mean:    {metrics.mean:.2f}")
print(f"Std:     {metrics.std:.2f}")
print(f"SNR:     {metrics.snr:.1f}")
print(f"Outliers: {metrics.outlier_fraction*100:.2f}%")
print(f"Saturated: {metrics.num_saturated_pixels}")

if metrics.warnings:
    print("Warnings:")
    for w in metrics.warnings:
        print(f"  - {w}")

Assessing In-Memory Arrays

from fits_tools import get_data_and_header
from fits_tools.quality_control import assess_quality_array

data, header = get_data_and_header("flat.fits")
metrics = assess_quality_array(data, header, frame_type="FLAT")
print(f"Grade: {metrics.quality_grade}")

Custom Thresholds

Quality assessment uses configurable thresholds. Create custom thresholds for your instrument:

from fits_tools.quality_control import QualityThresholds, assess_quality

custom = QualityThresholds(
    max_outlier_fraction=0.01,      # flag if > 1% outliers
    max_saturation_fraction=0.001,  # flag if > 0.1% saturated
    saturation_threshold=60000.0,   # ADU counted as saturated
    min_snr=50.0,                   # require SNR > 50
)

metrics = assess_quality("image.fits", thresholds=custom)

Calibration Quality

Measure how much calibration improved an image by comparing raw and calibrated versions:

from fits_tools.quality_control import assess_calibration_quality

result = assess_calibration_quality(
    "calibrated_science.fits",
    raw_file="raw_science.fits",
)

print(f"Grade: {result['quality_grade']} (passed: {result['passed']})")
improvement = result["improvement"]
print(f"Noise ratio (raw std / calibrated std): {improvement['noise_reduction']:.2f}")
if "snr_improvement" in improvement:
    print(f"SNR improvement: {improvement['snr_improvement']:.1f}x")

Batch Assessment

Assess quality across a set of files:

from fits_tools.quality_control import batch_quality_assessment
from pathlib import Path

files = sorted(Path("calibrated/").glob("*.fits"))
batch = batch_quality_assessment(files)

# Summary
print(f"Passed: {batch['passed']}/{batch['total_files']} "
      f"({batch['pass_rate']:.0%}); mean score "
      f"{batch['summary']['mean_quality_score']:.0f}")

# Per-file QualityMetrics are in batch["results"]
for m in batch["results"]:
    if not m.passed:
        print(f"  FAIL: {m.filepath} - {m.quality_grade} ({m.quality_score:.0f})")
        for e in m.errors:
            print(f"        {e}")

Quality Reports

Generate a formatted report from a batch assessment:

from fits_tools.quality_control import generate_quality_report

# Text report (returned as a string)
text = generate_quality_report(batch)
print(text)

# Save as JSON or CSV
generate_quality_report(batch, output_file="quality_report.json", fmt="json")
generate_quality_report(batch, output_file="quality_report.csv", fmt="csv")

Calibration Set Assessment

Assess the overall quality of a complete set of calibration frames:

from fits_tools.quality_control import assess_calibration_set_quality

bias_files = sorted(Path("bias_frames/").glob("*.fits"))
dark_files = sorted(Path("dark_frames/").glob("*.fits"))
flat_files = sorted(Path("flat_frames/").glob("*.fits"))

result = assess_calibration_set_quality(
    bias_files=bias_files,
    dark_files=dark_files,
    flat_files=flat_files,
)

print(f"Overall passed: {result['overall_passed']}")
for frame_type in ("bias", "dark", "flat"):
    summary = result[frame_type]
    print(f"{frame_type}: {summary['passed']}/{summary['total_files']} passed")
