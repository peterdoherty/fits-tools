The fits_tools.validate module provides report-oriented validation of FITS files. Unlike I/O functions that raise exceptions on errors, validation functions return a ValidationResult object describing what passed and what failed.

ValidationResult

All validation functions return a ValidationResult:

from fits_tools import validate_fits_file

result = validate_fits_file("image.fits")

if result.is_valid:
    print("File is valid")
else:
    for error in result.errors:
        print(f"ERROR: {error}")

for warning in result.warnings:
    print(f"WARNING: {warning}")

# Additional info
print(result.info)
FieldTypeDescription
is_validboolOverall pass/fail (False if any errors)
errorslist of strCritical validation failures
warningslist of strNon-fatal concerns
infodictSupplemental metadata

Basic File Validation

validate_fits_file

Checks basic FITS file integrity (can it be opened, does it have valid HDUs):

from fits_tools import validate_fits_file

result = validate_fits_file("image.fits")

quick_validate

A convenient one-stop check that applies sensible defaults for CCD/CMOS data:

from fits_tools import quick_validate

# General validation
result = quick_validate("image.fits")

# Validate as a specific frame type
result = quick_validate("bias.fits", frame_type="BIAS")

Header Validation

Required Keywords

from fits_tools.validate import validate_required_keywords

result = validate_required_keywords(
    "image.fits",
    required_keywords=["EXPTIME", "IMAGETYP", "GAIN", "RDNOISE"],
)
if not result.is_valid:
    print("Missing keywords:")
    for err in result.errors:
        print(f"  {err}")

Keyword Values

Validate that keyword values satisfy specific constraints:

from fits_tools.validate import validate_header_values

result = validate_header_values("image.fits", {
    "EXPTIME": lambda v: v > 0,               # must be positive
    "GAIN": (0.1, 10.0),                       # must be in range
    "IMAGETYP": {"LIGHT", "FLAT", "DARK"},     # must be one of these
})

Constraint types:

Keyword Types

from fits_tools.validate import validate_keyword_types

result = validate_keyword_types("image.fits", {
    "EXPTIME": float,
    "NAXIS1": int,
    "OBJECT": str,
})

Data Validation

Data Range

from fits_tools.validate import validate_data_range

# Check that all pixels are within a valid range
result = validate_data_range("image.fits", min_value=0, max_value=65535)

NaN Check

from fits_tools.validate import validate_no_nans

result = validate_no_nans("image.fits")
if not result.is_valid:
    print("Image contains NaN values!")

Image Dimensions

from fits_tools.validate import validate_image_dimensions

# Exact shape
result = validate_image_dimensions("image.fits", expected_shape=(2048, 2048))

# Minimum dimensions
result = validate_image_dimensions("image.fits", min_shape=(512, 512))

Frame-Type Validation

Validate that a file is a properly formed calibration frame:

from fits_tools.validate import (
    validate_bias_frame,
    validate_dark_frame,
    validate_flat_frame,
    validate_frame_type,
)

# Check that a file has a recognized frame type
result = validate_frame_type("image.fits")

# Validate as a bias frame (checks EXPTIME = 0, etc.)
result = validate_bias_frame("bias.fits")

# Validate as a dark frame (checks EXPTIME > 0)
result = validate_dark_frame("dark.fits", min_exptime=0.1)

# Validate as a flat frame (checks signal level)
result = validate_flat_frame("flat.fits", min_counts=1000, max_counts=50000)

Frame Set Validation

Validate a collection of files for consistency:

from fits_tools.validate import validate_frame_set

from pathlib import Path

bias_files = sorted(Path("bias_frames/").glob("*.fits"))
dark_files = sorted(Path("dark_frames/").glob("*.fits"))
flat_files = sorted(Path("flat_frames/").glob("*.fits"))

result = validate_frame_set(
    bias_files,
    check_dimensions=True,        # all files must have same shape
    check_frame_type=True,        # all files must have same IMAGETYP
    expected_frame_type="BIAS",
)

Calibration Set Validation

Validate an entire calibration data set (bias + dark + flat):

from fits_tools.validate import validate_calibration_set

result = validate_calibration_set(
    bias_files=bias_files,
    dark_files=dark_files,
    flat_files=flat_files,
)
if result.is_valid:
    print("Calibration set is ready")
else:
    for err in result.errors:
        print(f"  {err}")

Schema Validation

Validate a file against a schema dictionary that specifies required keywords, optional keywords, and constraints:

from fits_tools.validate import validate_against_schema

schema = {
    "required_keywords": ["EXPTIME", "IMAGETYP", "GAIN"],
    "keyword_types": {"EXPTIME": float, "NAXIS1": int},
    "keyword_constraints": {
        "EXPTIME": lambda v: v >= 0,
        "GAIN": (0.1, 100.0),
    },
    "min_shape": (100, 100),
}

result = validate_against_schema("image.fits", schema)

Array Validation

The fits_tools.array_validation module provides low-level validation for scalar values and NumPy arrays. These raise exceptions rather than returning results:

from fits_tools.array_validation import (
    validate_numeric,
    validate_positive,
    validate_shape,
    validate_shapes,
)

from fits_tools import get_data, get_header_value

image = get_data("image.fits")
bias = get_data("bias.fits")
flat = get_data("flat.fits")

# Validate a scalar parameter
exptime = validate_numeric(get_header_value("image.fits", "EXPTIME"), "exptime", minimum=0.0)

# Validate array shape
validate_shape(image, expected_ndim=2, operation="calibration")

# Validate that arrays have the same shape
validate_shapes(image, bias, flat, operation="calibration")
